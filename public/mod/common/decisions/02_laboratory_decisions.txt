namespace = laboratory

# Establish Magic Laboratory
establish_laboratory = {
	picture = "gfx/interface/illustrations/decisions/decision_laboratory.dds"
	desc = establish_laboratory_desc
	selection_tooltip = establish_laboratory_tooltip

	is_shown = {
		has_trait = mage
		mana >= 10
		location = { has_building = laboratory_building }
	}

	is_valid = {
		mana >= 10
		is_location_valid = yes
		has_capital = yes
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
	}

	effect = {
		add_mana = -10
		set_character_flag = laboratory_established
		capital_county = {
			add_county_modifier = {
				modifier = magical_laboratory
				years = 20
			}
		}
		add_character_modifier = {
			modifier = laboratory_owner
			years = 20
		}
	}

	ai_chance = {
		base = 30
		modifier = {
			add = 40
			has_trait = scholar
		}
	}
}
