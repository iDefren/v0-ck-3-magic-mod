namespace = ritual

# Ritual of Power
ritual_of_power = {
	picture = "gfx/interface/illustrations/decisions/decision_ritual.dds"
	desc = ritual_of_power_desc
	selection_tooltip = ritual_of_power_tooltip

	is_shown = {
		has_trait = mage
		mana >= 20
	}

	is_valid = {
		mana >= 20
		is_location_valid = yes
		NOT = { has_character_flag = performing_ritual }
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
	}

	effect = {
		set_character_flag = performing_ritual
		add_mana = -20
		add_personal_scheme_power = {
			scheme = active_schemes
			power = 30
		}
		create_character_event = { on_action = ritual_complete }
	}

	ai_chance = {
		base = 20
		modifier = {
			add = 30
			has_trait = ambitious
		}
		modifier = {
			add = 20
			has_trait = scholar
		}
	}
}

# Summon Familiar
summon_familiar = {
	picture = "gfx/interface/illustrations/decisions/decision_familiar.dds"
	desc = summon_familiar_desc
	selection_tooltip = summon_familiar_tooltip

	is_shown = {
		has_trait = mage
		mana >= 15
		NOT = { has_character_flag = has_familiar }
	}

	is_valid = {
		mana >= 15
		is_location_valid = yes
		learning >= 10
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
	}

	effect = {
		add_mana = -15
		set_character_flag = has_familiar
		if = {
			limit = { has_trait = demon }
			set_character_flag = demonic_familiar
		}
		else_if = {
			limit = { has_trait = angel }
			set_character_flag = celestial_familiar
		}
		else = {
			set_character_flag = arcane_familiar
		}
		custom_tooltip = summon_familiar_effect_tooltip
	}

	ai_chance = {
		base = 25
		modifier = {
			add = 40
			has_trait = scholar
		}
	}
}

# Bind Contract
bind_contract = {
	picture = "gfx/interface/illustrations/decisions/decision_contract.dds"
	desc = bind_contract_desc
	selection_tooltip = bind_contract_tooltip

	is_shown = {
		has_trait = mage
		mana >= 20
		OR = {
			has_trait = demon
			has_trait = angel
		}
	}

	is_valid = {
		mana >= 20
		is_location_valid = yes
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
	}

	effect = {
		add_mana = -20
		set_character_flag = contract_bound
		if = {
			limit = { has_trait = demon }
			add_trait = ambition_realized
			add_prestige = 50
		}
		else_if = {
			limit = { has_trait = angel }
			add_piety = 75
			add_prestige = 40
		}
	}

	ai_chance = {
		base = 30
		modifier = {
			add = 50
			has_trait = ambitious
		}
	}
}
