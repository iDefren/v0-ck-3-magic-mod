ritual_of_power = {
	title = ritual_of_power_title
	desc = ritual_of_power_desc
	picture = "gfx/interface/illustrations/decisions/decision_ritual.dds"
	selection_tooltip = ritual_of_power_tooltip

	is_shown = {
		OR = {
			has_trait = mage_initiate

			has_trait = mage_journeyman
			has_trait = mage_expert
			has_trait = mage_master

		}
		has_variable = arcane_mana
		var:arcane_mana >= 20
	}

	is_valid = {
		var:arcane_mana >= 20
		is_adult = yes
		is_capable_adult = yes
		NOT = { has_character_flag = performing_ritual }
	}

	effect = {
		set_character_flag = {
			flag = performing_ritual
			days = 30
		}
		change_variable = { name = arcane_mana add = -20 }
		add_prestige = 50
		trigger_event = ritual_events.0001
	}

	ai_check_interval = 36
	ai_potential = {
		has_trait = ambitious
	}
}

# Summon Familiar
summon_familiar = {
	title = summon_familiar_title
	desc = summon_familiar_desc
	picture = "gfx/interface/illustrations/decisions/decision_familiar.dds"
	selection_tooltip = summon_familiar_tooltip

	is_shown = {
		OR = {
			has_trait = mage_adept
			has_trait = mage_journeyman
			has_trait = mage_expert
			has_trait = mage_master
			has_trait = archmage
		}
		has_variable = arcane_mana
		var:arcane_mana >= 15
		NOT = { has_character_flag = has_familiar }
	}

	is_valid = {
		var:arcane_mana >= 15
		learning >= 10
		is_capable_adult = yes
	}

	effect = {
		change_variable = { name = arcane_mana add = -15 }
		set_character_flag = has_familiar
		if = {
			limit = {
				OR = {
					has_trait = demon_low
					has_trait = demon_mid
					has_trait = demon_high
					has_trait = demon_legendary
				}
			}
			set_character_flag = demonic_familiar
		}
		else_if = {
			limit = {
				OR = {
					has_trait = angel_lesser
					has_trait = angel_greater
					has_trait = angel_archangel
					has_trait = celestial_ascended
				}
			}
			set_character_flag = celestial_familiar
		}
		else = {
			set_character_flag = arcane_familiar
		}
		trigger_event = ritual_events.0010
	}

	ai_check_interval = 48
	ai_potential = {
		has_trait = scholar
	}
}
