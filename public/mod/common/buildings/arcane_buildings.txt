magic_laboratory = {
	construction_time = 730  # 2 years

	type = special
	slot = special_building

	character_modifier = {
		monthly_arcane_xp = 2
		learning = 1
		arcane_mana_regen = 5
	}

	county_modifier = {
		development_growth_factor = 0.05
	}

	cost = {
		gold = 400
		prestige = 200
	}

	is_shown = {
		holder = {
			is_any_mage = yes
		}
	}

	can_construct = {
		holder = {
			is_any_mage = yes
			learning >= 10
		}
	}

	ai_value = {
		value = 100
		if = {
			limit = { holder = { has_trait = mage_master } }
			add = 100
		}
		if = {
			limit = { holder = { has_trait = archmage } }
			add = 150
		}
	}

	next_building = magic_laboratory_tier_2

	flag = magical_building
}
