# ============================================================
# ARCANE DOMINION - common/on_action/arcane_on_actions.txt
# Hooks into CK3 on_actions for game events
# ============================================================

# ============================================================
# MONTHLY MANA REGENERATION
# Using random_events for monthly mana update
# ============================================================

on_monthly = {
	random_events = {
		chance_to_happen = 100
		100 = arcane.0600
	}
}

# ============================================================
# TRAIT GAIN - Initialize mana system
# ============================================================

on_trait_gained = {
	effect = {
		if = {
			limit = {
				OR = {
					scope:trait = trait:arcane_spark
					scope:trait = trait:mage_initiate
					scope:trait = trait:demon_low
					scope:trait = trait:angel_lesser
				}
				NOT = { has_variable = arcane_mana }
			}
			set_variable = {
				name = arcane_mana
				value = 0
			}
			set_variable = {
				name = arcane_mana_max
				value = 100
			}
			set_variable = {
				name = arcane_xp
				value = 0
			}
			# Trigger awakening event
			trigger_event = arcane.0001
		}
	}
}

# ============================================================
# YEARLY MAGICAL EVENTS
# ============================================================

on_yearly_pulse = {
	random_events = {
		chance_to_happen = 15
		100 = arcane.0800
	}
}
