# ============================================================
# DUEL BRACKET
# ============================================================
tournament.0010 = {
	type = character_event
	title = tournament.0010.t
	desc = tournament.0010.desc
	picture = "gfx/event_illustrations/tournament_duel.dds"

	is_triggered_only = yes

	option = {
		name = tournament.0010.a  # Channel magic for advantage
		trigger = {
			OR = {
				has_trait = mage_expert
				has_trait = mage_master
				has_trait = archmage
			}
			var:arcane_mana >= 40
		}
		spend_mana_40 = yes
		# Success based on prowess + magic experience
		random_list = {
			40 = {
				modifier = {
					add = 20
					has_trait = mage_master
				}
				modifier = {
					add = 30
					has_trait = archmage
				}
				modifier = {
					add = 15
					has_trait = mage_expert
				}
				trigger_event = tournament.0011  # Win
			}
			60 = {
				trigger_event = tournament.0012  # Lose
			}
		}
		gain_arcane_xp = { VALUE = 20 }
	}
	option = {
		name = tournament.0010.b  # Fight with skill alone
		trigger = { prowess >= 16 }
		random_list = {
			55 = { trigger_event = tournament.0011 }
			45 = { trigger_event = tournament.0012 }
		}
	}
}

	}

		add_piety = 200

	}
