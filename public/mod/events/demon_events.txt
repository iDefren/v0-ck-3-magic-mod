# ============================================================
# DEMON 0100-0120 - CORRUPTION FOCUS EVENTS
# ============================================================
demon.0100 = {
	type = character_event
	title = demon.0100.t
	desc = demon.0100.desc
	picture = "gfx/event_illustrations/soul_harvest.dds"

	is_triggered_only = yes

	option = {
		name = demon.0100.a  # Harvest a courtier's soul
		trigger = {
			any_court_character = {
				NOT = { has_trait = angel_lesser }
				NOT = { has_trait = demon_low }
			}
		}
		random_court_character = {
			limit = {
				NOT = { has_trait = angel_lesser }
				NOT = { has_trait = demon_low }
				NOT = { has_trait = bound_by_contract }
			}
			save_scope_as = soul_target
			add_trait = soul_tainted
			add_opinion = {
				modifier = horrified_opinion
				target = prev
				opinion = -40
			}
		}
		gain_mana_50 = yes
		gain_arcane_xp = { VALUE = 30 }
	}
	option = {
		name = demon.0100.b  # Harvest from a prisoner
		trigger = { any_prisoner = { always = yes } }
		random_prisoner = {
			add_trait = soul_tainted
		}
		gain_mana_100 = yes
		gain_arcane_xp = { VALUE = 20 }
	}
}

# DEMON 0700 - CONTRACT OFFER DECISION
# ============================================================
demon.0700 = {
	type = character_event
	title = demon.0700.t
	desc = demon.0700.desc
	picture = "gfx/event_illustrations/contract_offer.dds"

	is_triggered_only = yes

	option = {
		name = demon.0700.a  # Choose a target
		trigger = {
			any_known_character = {
				NOT = { has_trait = bound_by_contract }
				NOT = { has_trait = angel_lesser }
			}
		}
		random_known_character = {
			limit = {
				NOT = { has_trait = bound_by_contract }
				NOT = { has_trait = angel_lesser }
				NOT = { has_trait = demon_low }
				learning >= 6
			}
			save_scope_as = contract_target
		}
		trigger_event = demon.0701
	}
}

demon.0701 = {
