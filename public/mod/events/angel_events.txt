angel.0031 = {
	type = character_event
	title = angel.0031.t
	desc = angel.0031.desc
	picture = "gfx/event_illustrations/celestial_known.dds"

	is_triggered_only = yes

	option = {
		name = angel.0031.a
		# Demons worldwide gain negative opinion
		every_known_character = {
			limit = {
				OR = {
					has_trait = demon_mid
					has_trait = demon_high
					has_trait = demon_legendary
				}
			}
			add_opinion = {
				modifier = celestial_enemy
				target = prev
				opinion = -80
			}
		}
		add_prestige = 500
	}
}

# ANGEL 0500-0520 - DIVINE SOVEREIGNTY TREE EVENTS
# ============================================================
angel.0500 = {
	type = character_event
	title = angel.0500.t
	desc = angel.0500.desc

	is_triggered_only = yes

	option = {
		name = angel.0500.a  # Bless a vassal
		trigger = { any_vassal = { always = yes } }
		random_vassal = {
			save_scope_as = blessed_target
			add_character_modifier = {
				modifier = celestial_blessed
				months = 24
			}
			add_piety = 500
		}
		gain_arcane_xp = { VALUE = 30 }
		add_piety = 300
	}
	option = {
		name = angel.0500.b  # Bless a child
		trigger = { any_child = { always = yes } }
		random_child = {
			add_character_modifier = {
				modifier = celestial_blessed
				months = 36
			}
			add_trait = half_angel
		}
		gain_arcane_xp = { VALUE = 40 }
	}
}

# ============================================================
# ANGEL 0700 - BLESSING DECISION EVENT
# ============================================================
angel.0700 = {
	type = character_event
	title = angel.0700.t
	desc = angel.0700.desc
	picture = "gfx/event_illustrations/angel_blessing_choice.dds"

	is_triggered_only = yes

	option = {
		name = angel.0700.a  # Bless a specific character
		random_known_character = {
			limit = {
				NOT = { has_character_modifier = celestial_blessed }
				is_alive = yes
			}
			save_scope_as = bless_target
			add_character_modifier = {
				modifier = celestial_blessed
				months = 24
			}
		}
		spend_mana_40 = yes
		gain_arcane_xp = { VALUE = 20 }
		add_piety = 200
	}
	option = {
		name = angel.0700.b  # Bless an entire county
		trigger = { prestige >= 500 }
		capital_county = {
			add_county_modifier = {
				modifier = minor_celestial_blessing
				years = 5
			}
		}
		spend_mana_80 = yes
		gain_arcane_xp = { VALUE = 40 }
		add_piety = 500
	}
}
