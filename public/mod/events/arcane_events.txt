# ============================================================
# EVENT 0001 - RANDOM AWAKENING (rare for non-mages)
# ============================================================
arcane.0001 = {
	type = character_event
	title = arcane.0001.t
	desc = arcane.0001.desc
	picture = "gfx/event_illustrations/arcane_awakening.dds"

	trigger = {
		NOT = { has_trait = arcane_spark }
		NOT = { has_trait = mage_initiate }
		NOT = { has_trait = mundane_soul }
		learning >= 6
		random <= 3  # Very rare 3% chance per year
	}

	option = {
		name = arcane.0001.a
		desc = arcane.0001.a.desc
		add_trait = arcane_spark
		initialize_mana_system = yes
		gain_arcane_xp = { VALUE = 10 }
		show_as_tooltip = { add_trait = arcane_spark }
	}

	option = {
		name = arcane.0001.b
		desc = arcane.0001.b.desc
		add_trait = mundane_soul
		add_character_modifier = {
			modifier = mana_starved
			years = 2
		}
	}
}

arcane.0051 = {
	type = character_event
	title = arcane.0051.t
	desc = arcane.0051.desc
	picture = "gfx/event_illustrations/seek_teacher.dds"

	is_triggered_only = yes

	# Teacher found in court?
	trigger = {
		any_court_character = {
			OR = {
				has_trait = mage_master
				has_trait = archmage
			}
		}
	}

	option = {
		name = arcane.0051.a  # Accept teacher
		random_court_character = {
			limit = {
				OR = {
					has_trait = mage_master
					has_trait = archmage
				}
			}
			save_scope_as = teacher_mage
		}
		add_character_modifier = {
			modifier = student_of_magic
			years = 3
		}
		gain_arcane_xp = { VALUE = 50 }
		scope:teacher_mage = {
			add_trait_xp = {
				trait = arcane_teacher
				value = 20
			}
		}
	}
	option = {
		name = arcane.0051.b  # Study alone
		gain_arcane_xp = { VALUE = 20 }
	}
}

# ============================================================
# EVENT 9000 - MAGICAL CRISIS CHECK
# ============================================================
arcane.9000 = {
	type = character_event
	hidden = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		OR = {
			has_trait = archmage
			has_trait = demon_legendary
			has_trait = celestial_ascended
		}
	}

	# Trigger magical crisis if world has too much arcane power
	immediate = {
		if = {
			limit = {
				global_var:total_arcane_power >= 5000
				NOT = { has_global_flag = magical_crisis_active }
			}
			set_global_flag = magical_crisis_active
			trigger_event = crisis_events.0001
		}
	}
}
