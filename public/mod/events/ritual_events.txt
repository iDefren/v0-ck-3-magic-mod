# ============================================================
# AWAKENING RITUAL (Non-mage attempts to gain magic)
# ============================================================
ritual_events.0001 = {
	type = character_event
	title = ritual_events.0001.t
	desc = ritual_events.0001.desc
	picture = "gfx/event_illustrations/awakening_ritual.dds"

	is_triggered_only = yes

	immediate = {
		add_gold = -200
	}

	random_list = {
		25 = {
			# Full success - awakened as mage
			modifier = { factor = 2 learning >= 15 }
			modifier = { factor = 1.5 any_close_family_member = { is_any_mage = yes } }
			option = {
				name = ritual_events.0001.a
				add_trait = arcane_spark
				initialize_mana_system = yes
				gain_arcane_xp = { VALUE = 50 }
				trigger_event = ritual_events.0002
			}
		}
		35 = {
			# Partial success - magical sensitivity
			modifier = { factor = 1.5 learning >= 12 }
			option = {
				name = ritual_events.0001.b
				add_character_modifier = {
					modifier = magical_sensitivity
					years = 5
				}
				add_learning_skill = 2
			}
		}
		30 = {
			# Failure - nothing happens
			option = {
				name = ritual_events.0001.c
				stress_impact = { base = 20 }
				add_prestige = -50
			}
		}
		10 = {
			# Catastrophic failure
			option = {
				name = ritual_events.0001.d
				add_trait = mundane_soul
				health_impact_add = -1.0
				stress_impact = { base = 50 }
			}
		}
	}
}

ritual_events.0102 = {
	type = character_event
	title = ritual_events.0102.t
	desc = ritual_events.0102.desc
	picture = "gfx/event_illustrations/phylactery.dds"

	is_triggered_only = yes

	option = {
		name = ritual_events.0102.a  # Create phylactery
		trigger = {
			any_prisoner = { always = yes }
			gold >= 500
		}
		add_gold = -500
		random_prisoner = {
			death = { death_reason = death_sacrificed }
		}
		add_trait = phylactery_bound
		add_trait = lich_touched
		remove_character_modifier = immortality_quest
		gain_arcane_xp = { VALUE = 200 }
		add_piety = -2000
		add_dread = 80
	}

	option = {
		name = ritual_events.0102.b  # Not ready
		change_variable = {
			name = immortality_progress
			add = 20
		}
		trigger_event = { id = ritual_events.0101 months = { 12 36 } }
	}
}

ritual_events.0201 = {
	type = character_event
	title = ritual_events.0201.t
	desc = ritual_events.0201.desc
	picture = "gfx/event_illustrations/blood_moon_power.dds"

	is_triggered_only = yes

	option = {
		name = ritual_events.0201.a  # Embrace dark power
		add_character_modifier = {
			modifier = blood_moon_empowered
			years = 5
		}
		if = {
			limit = { NOT = { is_any_demon = yes } }
			add_trait = demon_low
		}
		else_if = {
			limit = { has_trait = demon_low }
			remove_trait = demon_low
			add_trait = demon_mid
		}
		gain_arcane_xp = { VALUE = 100 }
		add_piety = -1000
		add_dread = 50
		add_character_modifier = {
			modifier = blood_moon_ritual_done
			years = 10
		}
	}
}
