# ============================================================
# LABORATORY UPGRADE
# ============================================================
laboratory_events.0001 = {
	type = character_event
	title = laboratory_events.0001.t
	desc = laboratory_events.0001.desc
	picture = "gfx/event_illustrations/laboratory.dds"

	is_triggered_only = yes

	immediate = {
		add_gold = -300
	}

	option = {
		name = laboratory_events.0001.a  # Upgrade successful
		trigger = { learning >= 15 }
		capital_county = {
			if = {
				limit = { has_building = magic_laboratory }
				remove_building = magic_laboratory
				add_building = magic_laboratory_tier_2
			}
			else_if = {
				limit = { has_building = magic_laboratory_tier_2 }
				remove_building = magic_laboratory_tier_2
				add_building = magic_laboratory_tier_3
			}
			else_if = {
				limit = { has_building = magic_laboratory_tier_3 }
				remove_building = magic_laboratory_tier_3
				add_building = magic_laboratory_tier_4
			}
		}
		gain_arcane_xp = { VALUE = 40 }
		add_prestige = 100
	}

	option = {
		name = laboratory_events.0001.b  # Failed upgrade
		trigger = { learning < 15 }
		add_gold = -100
		stress_impact = { base = 15 }
	}
}

laboratory_events.0400 = {
	type = character_event
	title = laboratory_events.0400.t
	desc = laboratory_events.0400.desc
	picture = "gfx/event_illustrations/train_apprentice.dds"

	trigger = {
		has_magic_laboratory = yes
		is_master_level_mage = yes
		any_courtier = {
			can_learn_magic = yes
		}
	}

	mean_time_to_happen = {
		years = 3
	}

	option = {
		name = laboratory_events.0400.a  # Take an apprentice
		random_courtier = {
			limit = { can_learn_magic = yes }
			save_scope_as = apprentice
		}
		scope:apprentice = {
			add_trait = arcane_spark
			initialize_mana_system = yes
			add_character_modifier = {
				modifier = student_of_magic
				years = 5
			}
			add_opinion = {
				target = root
				modifier = grateful_teacher
				years = -1
			}
		}
		add_trait = arcane_teacher
		gain_arcane_xp = { VALUE = 30 }
	}

	option = {
		name = laboratory_events.0400.b  # No time for students
		add_prestige = -20
	}
}
