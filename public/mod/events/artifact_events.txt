artifact_events.0050 = {
	type = character_event
	title = artifact_events.0050.t
	desc = artifact_events.0050.desc
	picture = "gfx/event_illustrations/enhance_artifact.dds"

	is_triggered_only = yes

	immediate = {
		spend_mana_80 = yes
		add_gold = -400
	}

	option = {
		name = artifact_events.0050.a  # Enhance with fire
		trigger = { has_trait = spec_pyromancer }
		random_artifact = {
			limit = { has_artifact_modifier = arcane_artifact }
			add_artifact_modifier = fire_enchanted
		}
		gain_arcane_xp = { VALUE = 50 }
	}

	option = {
		name = artifact_events.0050.b  # Enhance with mana
		random_artifact = {
			limit = { has_artifact_modifier = arcane_artifact }
			add_artifact_modifier = mana_battery
		}
		gain_arcane_xp = { VALUE = 45 }
	}

	option = {
		name = artifact_events.0050.c  # Enhance with soul power
		trigger = {
			has_dark_alignment = yes
			any_prisoner = { always = yes }
		}
		random_prisoner = {
			death = { death_reason = death_sacrificed }
		}
		random_artifact = {
			limit = { has_artifact_modifier = arcane_artifact }
			add_artifact_modifier = soul_hungry
		}
		gain_arcane_xp = { VALUE = 70 }
		add_piety = -400
	}
}

# ============================================================
# ARTIFACT FEEDING (Soul Hungry artifacts)
# ============================================================
artifact_events.0100 = {
	type = character_event
	title = artifact_events.0100.t
	desc = artifact_events.0100.desc
	picture = "gfx/event_illustrations/soul_feed.dds"

	is_triggered_only = yes

	option = {
		name = artifact_events.0100.a  # Feed a prisoner
		random_prisoner = {
			death = { death_reason = death_sacrificed }
		}
		random_artifact = {
			limit = { has_artifact_modifier = soul_hungry }
			add_artifact_modifier = soul_sated
		}
		add_character_modifier = {
			modifier = soul_artifact_empowered
			years = 2
		}
		gain_arcane_xp = { VALUE = 30 }
		add_piety = -200
		add_dread = 20
	}

	option = {
		name = artifact_events.0100.b  # Refuse to feed
		random_artifact = {
			limit = { has_artifact_modifier = soul_hungry }
			add_artifact_modifier = soul_starving
		}
		# Artifact may turn on wielder
		trigger_event = { id = artifact_events.0101 days = { 30 180 } }
	}
}

artifact_events.0101 = {
	type = character_event
	title = artifact_events.0101.t
	desc = artifact_events.0101.desc
	picture = "gfx/event_illustrations/hungry_artifact.dds"

	}
}

artifact_events.0201 = {
	type = character_event
	title = artifact_events.0201.t
	desc = artifact_events.0201.desc
	picture = "gfx/event_illustrations/identify_artifact.dds"

	is_triggered_only = yes

	random_list = {
		40 = {
			# Beneficial
			option = {
				name = artifact_events.0201.a

					remove_artifact_modifier = unknown_enchant

				name = artifact_events.0201.b

					remove_artifact_modifier = unknown_enchant
					add_artifact_modifier = soul_hungry
				}

		}
		30 = {
			# Actually cursed
			option = {
				name = artifact_events.0201.c

					years = 3
				}
			}

# LEGENDARY ARTIFACT EVENTS

artifact_events.0300 = {
	type = character_event
	title = artifact_events.0300.t
	desc = artifact_events.0300.desc
	picture = "gfx/event_illustrations/legendary_artifact.dds"

	trigger = {
		has_trait = archmage
		has_advanced_laboratory = yes
		var:arcane_mana >= 200
		NOT = { has_character_modifier = legendary_crafting_done }
	}

	mean_time_to_happen = {
		years = 20
	}

	option = {
		name = artifact_events.0300.a  # Attempt legendary crafting
		trigger = { gold >= 2000 }
		add_gold = -2000
		spend_mana_200 = yes
		trigger_event = artifact_events.0301
	}

	option = {
		name = artifact_events.0300.b  # Not ready
		add_character_modifier = {
			modifier = legendary_crafting_done
			years = 5
		}
	}
}

artifact_events.0301 = {
	type = character_event
	title = artifact_events.0301.t
	desc = artifact_events.0301.desc
	picture = "gfx/event_illustrations/create_legendary.dds"

	is_triggered_only = yes

	random_list = {
		40 = {
			modifier = { factor = 1.5 learning >= 28 }
			# Success
			option = {
				name = artifact_events.0301.a
				create_artifact = {
					name = artifact_legendary_masterpiece
					type = trinket
					modifier = legendary_arcane_artifact
					modifier = arcane_artifact
					modifier = mana_battery
					modifier = spell_amplifier
					quality = legendary
					save_scope_as = legendary_item
				}
				gain_arcane_xp = { VALUE = 200 }
				add_prestige = 1000
				add_character_modifier = {
					modifier = legendary_crafting_done
					years = -1
				}
			}
		}
		35 = {
			# Partial success
			option = {
				name = artifact_events.0301.b
				create_artifact = {
					name = artifact_powerful_item
					type = trinket
					modifier = arcane_artifact
					modifier = mana_battery
					quality = masterwork
					save_scope_as = craft_result
				}
				gain_arcane_xp = { VALUE = 80 }
				add_character_modifier = {
					modifier = legendary_crafting_done
					years = 10
				}
			}
		}
		25 = {
			# Catastrophic failure
			option = {
				name = artifact_events.0301.c
				health_impact_add = -2.0
				add_character_modifier = {
					modifier = arcane_overload
					months = 12
				}
				add_character_modifier = {
					modifier = legendary_crafting_done
					years = 15
				}
			}
		}
	}
}
