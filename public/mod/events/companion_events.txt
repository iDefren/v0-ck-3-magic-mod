# PEGASUS HUNT
# ============================================================
companion.0010 = {
	type = character_event
	title = companion.0010.t
	desc = companion.0010.desc
	picture = "gfx/event_illustrations/pegasus.dds"

	is_triggered_only = yes

	option = {
		name = companion.0010.a  # Approach with magic
		trigger = { var:arcane_mana >= 50 }
		spend_mana_50 = yes
		random_list = {
			60 = { trigger_event = companion.0011 }  # Success
			40 = { trigger_event = companion.0012 }  # Fail
		}
	}
	option = {
		name = companion.0010.b  # Use martial skill
		trigger = { prowess >= 14 }
		random_list = {
			40 = { trigger_event = companion.0011 }
			60 = { trigger_event = companion.0012 }
		}
	}
	option = {
		name = companion.0010.c  # Abandon the hunt
		remove_character_modifier = companion_hunt_active
	}
}

# ============================================================
# RANDOM COMPANION FIND EVENT (very rare, no hunt needed)
# ============================================================
companion.0200 = {
	type = character_event
	title = companion.0200.t
	desc = companion.0200.desc
	picture = "gfx/event_illustrations/unexpected_companion.dds"

	trigger = {
		NOT = { has_character_modifier = has_companion }
		OR = {
			has_trait = mage_adept
			has_trait = mage_journeyman
			has_trait = mage_expert
			has_trait = mage_master
			has_trait = archmage
		}
		random <= 1  # 1% chance per year
	}

	option = {
		name = companion.0200.a  # Accept it
		add_character_modifier = {
			modifier = has_companion
			years = -1
		}
		random_list = {
			40 = { add_character_modifier = { modifier = familiar_companion years = -1 } }
			30 = { add_character_modifier = { modifier = pegasus_companion years = -1 } }
			20 = { add_character_modifier = { modifier = phoenix_companion years = -1 } }
			10 = { add_character_modifier = { modifier = cerberus_companion years = -1 } }
		}
		gain_arcane_xp = { VALUE = 30 }
		add_prestige = 100
	}
	option = {
		name = companion.0200.b  # Turn it away
		custom_tooltip = companion.0200.b.tt
	}
}
