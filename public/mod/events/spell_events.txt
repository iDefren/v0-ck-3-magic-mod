# ============================================================
# FIREBALL SPELL (Pyromancer)
# ============================================================
spell_events.0001 = {
	type = character_event
	title = spell_events.0001.t
	desc = spell_events.0001.desc
	picture = "gfx/event_illustrations/fireball.dds"

	is_triggered_only = yes

	immediate = {
		spend_mana_25 = yes
	}

	# Target selection
	option = {
		name = spell_events.0001.a  # Target enemy army
		trigger = {
			any_war_enemy = { always = yes }
		}
		random_war_enemy = {
			save_scope_as = fireball_target
		}
		trigger_event = spell_events.0002
	}

	option = {
		name = spell_events.0001.b  # Target a rival
		trigger = {
			any_relation = {
				type = rival
				always = yes
			}
		}
		random_relation = {
			type = rival
			save_scope_as = fireball_target
		}
		trigger_event = spell_events.0002
	}

	option = {
		name = spell_events.0001.c  # Display of power (prestige)
		add_prestige = 100
		gain_arcane_xp = { VALUE = 10 }
		custom_tooltip = spell_events.0001.c.tt
	}
}

spell_events.0002 = {
	type = character_event
	title = spell_events.0002.t
	desc = spell_events.0002.desc
	picture = "gfx/event_illustrations/fireball_impact.dds"

	is_triggered_only = yes

	# Success based on fire magic power and learning
	option = {
		name = spell_events.0002.a  # Direct hit!
		trigger = {
			learning >= 12
			OR = {
				has_trait = spec_pyromancer
				has_trait = archmage
			}
		}
		scope:fireball_target = {
			health_impact_add = -2.0
			add_character_modifier = {
				modifier = badly_burned
				months = 12
			}
		}
		gain_arcane_xp = { VALUE = 25 }
		add_prestige = 50
	}

	option = {
		name = spell_events.0002.b  # Partial success
		trigger = { learning >= 8 }
		scope:fireball_target = {
			health_impact_add = -0.8
			add_character_modifier = {
				modifier = minor_burns
				months = 6
			}
		}
		gain_arcane_xp = { VALUE = 15 }
	}

	option = {
		name = spell_events.0002.c  # Failure
		trigger = { learning < 8 }
		add_character_modifier = {
			modifier = spell_backfire
			months = 3
		}
		health_impact_add = -0.5
	}
}

# HEALING LIGHT (Healer)
# ============================================================
spell_events.0050 = {
	type = character_event
	title = spell_events.0050.t
	desc = spell_events.0050.desc
	picture = "gfx/event_illustrations/healing_light.dds"

	is_triggered_only = yes

	immediate = {
		spend_mana_25 = yes
	}

	option = {
		name = spell_events.0050.a  # Heal yourself
		health_impact_add = 2.0
		if = {
			limit = { has_trait = wounded }
			remove_trait = wounded
		}
		if = {
			limit = { has_trait = ill }
			remove_trait = ill
		}
		gain_arcane_xp = { VALUE = 15 }
	}

	option = {
		name = spell_events.0050.b  # Heal a family member
		trigger = {
			any_close_family_member = {
				OR = {
					has_trait = wounded
					has_trait = ill
					health < 3
				}
			}
		}
		random_close_family_member = {
			limit = {
				OR = {
					has_trait = wounded
					has_trait = ill
					health < 3
				}
			}
			health_impact_add = 2.0
			remove_trait = wounded
			remove_trait = ill
		}
		gain_arcane_xp = { VALUE = 20 }
		add_piety = 50
	}

	option = {
		name = spell_events.0050.c  # Heal a courtier (reputation)
		random_courtier = {
			limit = {
				OR = {
					has_trait = wounded
					has_trait = ill
				}
			}
			health_impact_add = 1.5
			remove_trait = wounded
			remove_trait = ill
			add_opinion = {
				target = root
				modifier = healed_by_magic
				years = 10
			}
		}
		add_prestige = 50
		add_piety = 100
		gain_arcane_xp = { VALUE = 25 }
	}
}
